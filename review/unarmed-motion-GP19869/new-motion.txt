          else {
            // Starting/stopping fades the swing, including held beats used by the camera tool.
            POSE.unarmedMove = _ease(POSE.unarmedMove || 0, (moving || GAIT.holdU != null) && grounded ? 1 : 0, dt, 10);
            const move = POSE.unarmedMove, runMix = _clamp01(POSE.gaitRun || 0);
            // Studio foot contacts have a different phase from the fallback sine gait. Blend that offset
            // with the same Studio/crouch weight as the legs; neither knee bend nor terrain can change it.
            const phase = walkPhase - (1.65 - 0.9 * runMix) * (POSE.gaitW || 0);
            const uL = Math.sin(phase), uR = -uL, quiet = 1 - 0.4 * POSE.crouch;
            const swing = (u) => u * (0.36 + 0.25 * runMix) * quiet;
            const swayL = Math.sin(t * 1.1) * 0.02, swayR = Math.sin(t * 1.1 + 0.7) * 0.02;
            swL = swayL + (swing(uL) - swayL) * move;
            swR = swayR + (swing(uR) - swayR) * move;
            // A small forearm lag softens the reversal; the bend closes gradually with the run weight.
            const flex = Math.sin(phase - 0.2), base = -0.28 - 0.78 * runMix, curl = 0.09 + 0.12 * runMix;
            bendL += (base - flex * curl - bendL) * move;
            bendR += (base + flex * curl - bendR) * move;
            pitch += (-0.02 + 0.1 * runMix - pitch) * move;
            out = Math.sin(t * 0.7) * 0.006 * (1 - move) + (0.04 + 0.11 * runMix) * move;
            torsoTwist -= uL * (0.035 + 0.04 * runMix) * rw * quiet * move;
          }
