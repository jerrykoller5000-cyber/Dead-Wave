const sites=[];
const add=(id,p,r=14)=>{if(p&&Number.isFinite(p.x)&&Number.isFinite(p.z))sites.push({id,x:p.x,z:p.z,r});};
add('hq',{x:0,z:0},24);add('landing',{x:42,z:-20},9);
for(const key of ['campsites','cabins','sheds','wrecks','caves','bridges'])for(let i=0;i<(T.POI[key]||[]).length;i++)add(key+'-'+i,T.POI[key][i],key==='bridges'?9:14);
add('mast',T.POI.mast,20);add('graveyard',T.POI.graveyard);add('tower',T.POI.tower,12);add('dock',T.POI.dock,9);
const out=[];
for(const [id,a0,a1,target] of [['ne',0,90,45],['sw',180,270,225]])for(const r of [115,120,125,130,135,140,145,150]){
 const valid=[];
 for(let deg=a0;deg<=a1;deg++){
  const a=deg*Math.PI/180,x=r*Math.cos(a),z=r*Math.sin(a),nx=-Math.sin(a),nz=Math.cos(a);
  const wet=[-2.5,0,2.5].some(s=>T.waterDepthAt(x+nx*s,z+nz*s)>.05);
  const nearest=sites.map(p=>({id:p.id,clear:Math.hypot(x-p.x,z-p.z)-p.r-2.5})).sort((u,v)=>u.clear-v.clear)[0];
  const slope=T.terrainSlope(x,z);
  if(!wet&&nearest.clear>=15&&slope<=.3)valid.push({deg,x:+x.toFixed(1),z:+z.toFixed(1),clear:+nearest.clear.toFixed(1),slope:+slope.toFixed(2)});
 }
 valid.sort((a,b)=>Math.abs(a.deg-target)-Math.abs(b.deg-target));
 out.push({id,r,count:valid.length,best:valid.slice(0,6)});
}
return out;
