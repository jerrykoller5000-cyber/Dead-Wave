    function vaultClearAt(x,y,z,bodyHeight,ignore) {
      if(Math.hypot(x,z)>LAND_MAX-PLAYER_RADIUS && !training.active)return false;
      const ceiling=deckCeilingAt(x,z,y);
      if(ceiling!==null && ceiling<y+bodyHeight+.06)return false;
      for(const b of buildsNear(x,z,2,[])) {
        if(b===ignore||b.hp<=0||!b.mesh)continue;
        const p=pushOutOfBuild(b,x,z,PLAYER_RADIUS,true,y);
        if(Math.hypot(p.x-x,p.z-z)>.015)return false;
      }
      const p=resolveStaticProps(x,z,PLAYER_RADIUS,y);
      if(Math.hypot(p.x-x,p.z-z)>.015)return false;
      if(house.present) { const p=pushOutOfHouse(x,z,PLAYER_RADIUS);if(Math.hypot(p.x-x,p.z-z)>.015)return false; }
      if(training.active && training.tg) {
        const tg=training.tg;
        // Architectural, station and protective-fence solids remain solid during a vault.
        const blockers=tg.solids;
        for(const b of blockers)if(y<b.maxY && y+bodyHeight>b.minY &&
          x>b.minX-PLAYER_RADIUS && x<b.maxX+PLAYER_RADIUS && z>b.minZ-PLAYER_RADIUS && z<b.maxZ+PLAYER_RADIUS)return false;
      }
      return waterDepthAt(x,z)<.7;
    }

        reloading||swapT>0||knifeAnimT>0||grenadeCharging||grenadeAnimBusy()||playerSwimming||onAnyLadder()||

      const direction=speed>.65?{x:velX,z:velZ}:{x:Math.sin(aimYaw),z:Math.cos(aimYaw)};
      let best=null,bestDist=Infinity;
      for(const b of buildsNear(position.x,position.z,2.6,[])) {
        if(!pieceVaultable(b))continue;
        const box=thinBoxFor(b),baseY=b.mesh.position.y;
        const topY=baseY+(b.type==='wall'&&b.opening==='window'?1.0:pieceHeight(b.type));
        const plan=planVault({piece:b,box,position,direction,baseY,topY,speed,groundAt:playerGroundY,clearAt:vaultClearAt});
        if(!plan)continue;
        const d=Math.hypot(plan.x1-position.x,plan.z1-position.z);
        if(d<bestDist){best=plan;bestDist=d;}
      }
      if(!best)return false;
      best.exitSpeed=Math.min(speed,PHYS.walk);vault=best;vaultPoseFrame=sampleVault(best,0);
      velX=0;velZ=0;wishVX=0;wishVZ=0;vy=0;grounded=false;coyoteT=0;
      jumpBufferT=0;resetJump(jumpFeel);akimboHand=0;offhandMount.visible=false;
      if(!vaultCanFire(currentWeapon,unarmed))fireHeld=false;
      AudioSys.vault?.();return true;

    function updateVault(dt) {
      if(!vault)return;
      const v=vault;
      if(marineDown()||gameOver||scriptedKill||training.wake){vault=null;vaultPoseFrame=null;return;}
      v.t=vaultHoldU==null?Math.min(v.dur,v.t+dt):v.dur*vaultHoldU;
      const p=sampleVault(v);
      if(!vaultClearAt(p.x,p.y,p.z,p.bodyHeight,v.piece)) {
        player.position.set(v.x0,v.y0,v.z0);vault=null;vaultPoseFrame=null;grounded=true;vy=0;return;
      }
      player.position.set(p.x,p.y,p.z);vaultPoseFrame=p;vy=0;grounded=false;
      if(p.u>=1) {
        vault=null;vaultPoseFrame=null;grounded=true;coyoteT=.12;
        velX=v.nx*v.exitSpeed*.6;velZ=v.nz*v.exitSpeed*.6;wishVX=velX;wishVZ=velZ;
        landJump(jumpFeel,3.2);AudioSys.land?.(.45);
      }
    }
    function poseVault() {
      if(!vault||!vaultPoseFrame)return;
      const P=vaultPoseFrame,ud=marine.userData;
      const yaw=Math.atan2(vault.nx,vault.nz),turn=_wrapAngle(yaw-player.rotation.y);
      const k=_smooth(Math.min(1,P.u/.14))*(1-_smooth(Math.max(0,(P.u-.84)/.16)));
      const mix=(o,x,y,z)=>{o.rotation.x+=(x-o.rotation.x)*k;o.rotation.y+=(y-o.rotation.y)*k;o.rotation.z+=(z-o.rotation.z)*k;};
      mix(ud.lowerBody,0,turn,0);
      const aimTurn=Math.max(-1.15,Math.min(1.15,_wrapAngle(aimYaw-yaw)));
      mix(ud.torsoG,P.lean,aimTurn*.55,P.roll);
      ud.headG.rotation.x+=(-P.lean*.55-ud.headG.rotation.x)*k;
      marine.position.y+=(-ROLL_PIVOT_Y+P.dip-marine.position.y)*k;
      mix(ud.legLG,P.thighL,0,-.12*P.over);
      mix(ud.kneeLG,P.kneeL,0,0);mix(ud.ankleLG,-.35*P.lead,0,0);
      mix(ud.legRG,P.thighR,0,.14*P.over);
      mix(ud.kneeRG,P.kneeR,0,0);mix(ud.ankleRG,-.3*P.trail,0,0);
    }
