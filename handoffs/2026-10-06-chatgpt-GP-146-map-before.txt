    function bakeMinimap() {
      const ctx = mapBake.getContext('2d');
      const img = ctx.createImageData(BAKE_PX, BAKE_PX);
      const d = img.data;
      for (let py = 0; py < BAKE_PX; py++) {
        const wz = MAP_R - (py + 0.5) / bakeScale;
        for (let px = 0; px < BAKE_PX; px++) {
          const wx = MAP_R - (px + 0.5) / bakeScale;
          const i = (py * BAKE_PX + px) * 4;
          if (wx * wx + wz * wz > MAP_R * MAP_R) { d[i + 3] = 0; continue; }
          const h = sampleHeight(wx, wz);
          let r, g, b;
          if (waterLevelAt(wx, wz) !== null) {
            r = 38; g = 92; b = 150;
          } else if (h > 34) {
            const t = Math.min(1, (h - 34) / 40);
            r = 120 + t * 120; g = 120 + t * 122; b = 118 + t * 128;
          } else {
            const t = Math.max(0, Math.min(1, (h + 8) / 26));
            r = 58 + t * 46; g = 118 + t * 26; b = 58 + t * 8;
          }
          // A touch of hillshade so relief reads when it is blown up on the minimap.
          const hx = sampleHeight(wx - 1.5, wz) - h, hz = sampleHeight(wx, wz - 1.5) - h;
          const shade = Math.max(0.78, Math.min(1.18, 1 + (hx + hz) * 0.06));
          d[i] = r * shade; d[i + 1] = g * shade; d[i + 2] = b * shade; d[i + 3] = 235;
        }
      }
      ctx.putImageData(img, 0, 0);
      // Tracks are stroked rather than tested per pixel: at this resolution a per-pixel
      // distance-to-every-segment test cost more than the rest of the bake put together.
      ctx.save();
      ctx.beginPath();
      ctx.arc(BAKE_PX / 2, BAKE_PX / 2, MAP_R * bakeScale, 0, Math.PI * 2);
      ctx.clip();
      ctx.strokeStyle = 'rgb(156,138,104)'; // tan, not rust: must never be mistaken for a zombie blip
      ctx.lineJoin = 'round'; ctx.lineCap = 'round';
      for (const line of PATHS) {
        // Busier trails draw broader, as they are on the ground (at least the old
        // width, so a footpath still reads on the map).
        ctx.lineWidth = Math.max(3.8, (line.hw || TRAIL_HW) * 2.6) * bakeScale;
        ctx.beginPath();
        line.forEach((p, k) => { const q = worldToBake(p.x, p.z); if (k) ctx.lineTo(q.x, q.y); else ctx.moveTo(q.x, q.y); });
        ctx.stroke();
      }
      ctx.restore();
      // Static landmark markers.
      function mark(x, z, color) {
        const p = worldToBake(x, z);
        ctx.fillStyle = 'rgba(0,0,0,0.55)';
        ctx.fillRect(p.x - 6, p.y - 6, 12, 12);
        ctx.fillStyle = color;
        ctx.fillRect(p.x - 4.5, p.y - 4.5, 9, 9);
      }
      for (const c of POI.campsites) mark(c.x, c.z, '#ffb347');
      for (const c of POI.cabins) mark(c.x, c.z, '#c9a17c');
      if (POI.tower) mark(POI.tower.x, POI.tower.z, '#e6d27a');
      if (POI.graveyard) mark(POI.graveyard.x, POI.graveyard.z, '#b9b9c9');
      for (const b of POI.bridges) mark(b.x, b.z, '#d9c7a0');
      for (const s of POI.sheds) mark(s.x, s.z, '#9a7b5a');
      for (const w of POI.wrecks) mark(w.x, w.z, '#6a8aaa');
      if (POI.mast) mark(POI.mast.x, POI.mast.z, '#ff4466');
      if (POI.dock) mark(POI.dock.x, POI.dock.z, '#5a9ccc');
    }

      ctx.fill();
    }

      ctx.fillStyle = color;
      ctx.fillRect(p.x - s / 2, p.y - s / 2, s, s);
    }

      ctx.lineWidth = 3;
      ctx.strokeStyle = 'rgba(0,0,0,0.85)';
      ctx.stroke();
      ctx.fillStyle = '#ffffff';
      ctx.fill();
      ctx.restore();
    }

        return { x: C + lx * k, y: C - ly * k };
